          from pathlib import Path
          from datetime import date
          import re
          
          target = "https://cepoes.org/presupuesto/descentralizacion/"
          route = "/presupuesto/descentralizacion/"
          lastmod = date.today().isoformat()
          
          # --- sitemap.xml ---
          p = Path("deploy/site-overlay/sitemap.xml")
          text = p.read_text(encoding="utf-8")
          if "ns0:" in text or "xmlns:ns0=" in text:
              raise SystemExit("sitemap.xml conserva namespace ns0; requiere reparación previa controlada")
          if "</urlset>" not in text:
              raise SystemExit("sitemap.xml no tiene cierre </urlset>")
          pattern = re.compile(
              r"\s*<url>\s*<loc>" + re.escape(target) +
              r"</loc>\s*(?:<lastmod>[^<]+</lastmod>\s*)?</url>"
          )
          text = pattern.sub("", text)
          entry = f"  <url><loc>{target}</loc><lastmod>{lastmod}</lastmod></url>\n"
          text = text.replace("</urlset>", entry + "</urlset>", 1)
          if not text.endswith("\n"):
              text += "\n"
          p.write_text(text, encoding="utf-8")
          
          # --- sitemap.txt ---
          p = Path("deploy/site-overlay/sitemap.txt")
          lines = [x.strip() for x in p.read_text(encoding="utf-8").splitlines() if x.strip()]
          lines = [x for x in lines if x != target] + [target]
          if len(lines) != len(set(lines)):
              raise SystemExit("sitemap.txt contiene URLs duplicadas ajenas a Descentralización")
          p.write_text("\n".join(lines) + "\n", encoding="utf-8")
          
          # --- Página pública ---
          p = Path("deploy/site-overlay/presupuesto/descentralizacion/index.html")
          html = p.read_text(encoding="utf-8")
          html = re.sub(r'<header class="top">.*?</header>\s*', '', html, count=1, flags=re.S)
          if '<html lang="es">' in html:
              html = html.replace('<html lang="es">', '<html data-theme="light" lang="es">', 1)
          elif 'data-theme=' not in html:
              html = re.sub(r'<html\b', '<html data-theme="light"', html, count=1)
          site_css = '<link href="/assets/site.css?v=233" rel="stylesheet">'
          if '/assets/site.css' not in html:
              html = html.replace('</style>', '</style>\n' + site_css, 1)
          favicon = '<link href="/assets/favicon.svg" rel="icon" type="image/svg+xml">'
          if '/assets/favicon.svg' not in html:
              anchor = site_css if site_css in html else '</style>'
              html = html.replace(anchor, anchor + '\n' + favicon, 1)
          canonical = '<link rel="canonical" href="https://cepoes.org/presupuesto/descentralizacion/">'
          canon_pat = re.compile(r'<link\b(?=[^>]*\brel=["\']canonical["\'])[^>]*>', re.I)
          if canon_pat.search(html):
              html = canon_pat.sub(canonical, html, count=1)
          else:
              if '</head>' not in html:

              html = html.replace('</head>', canonical + '\n</head>', 1)
          common = '<script defer src="/assets/common.js?v=254"></script>'
          if '/assets/common.js' not in html:
              if '</body>' not in html:
                  raise SystemExit("Página Descentralización sin </body>")

          git add deploy/site-overlay/sitemap.xml \
                  deploy/site-overlay/sitemap.txt \
                  deploy/site-overlay/presupuesto/descentralizacion/index.html \
                  deploy/site-overlay/assets/common.js \

          fi

      - name: Publicar build ya auditado en Hostinger
        id: publish
        shell: bash
        run: |
          set -euo pipefail
          lftp -u "$SFTP_USER","$SFTP_PASSWORD" -p "$SFTP_PORT" "sftp://$SFTP_HOST" <<EOF
          set sftp:auto-confirm yes
          set cmd:fail-exit yes
          set net:max-retries 3
          set net:timeout 30
          mirror -R --verbose --parallel=2 --no-perms --no-umask _audit_site "$REMOTE_DIR"
          bye
          EOF
          echo "✓ publicación SFTP completada"

      - name: Smoke final en producción
        id: smoke
        shell: bash
        run: |
          set -euo pipefail
          Q="audit=${{ github.run_id }}-${{ github.run_attempt }}-$(date +%s)"
          CURL=(--retry 6 --retry-delay 5 --connect-timeout 15 --max-time 45 -fsS -H "Cache-Control: no-cache" -H "Pragma: no-cache")

          curl "${CURL[@]}" "https://cepoes.org/?$Q" >/dev/null
          curl "${CURL[@]}" "https://cepoes.org/presupuesto/?$Q" -o /tmp/hub.html
          curl "${CURL[@]}" "https://cepoes.org/presupuesto/descentralizacion/?$Q" -o /tmp/des.html
          curl "${CURL[@]}" "https://cepoes.org/assets/common.js?$Q" -o /tmp/common.js
          curl "${CURL[@]}" "https://cepoes.org/sitemap.xml?$Q" -o /tmp/sitemap.xml

          grep -q '¿Cuánto poder llegó realmente a las Comunas?' /tmp/des.html
          grep -q 'class="active" href="/presupuesto/descentralizacion/"' /tmp/des.html
          grep -q 'href="/presupuesto/descentralizacion/"' /tmp/hub.html
          grep -q '/assets/common.js' /tmp/des.html
          grep -q "x.url==='/presupuesto/descentralizacion/'" /tmp/common.js
          grep -q 'https://cepoes.org/presupuesto/descentralizacion/' /tmp/sitemap.xml

          for name in descentralizacion-comunas.json descentralizacion-transparencia-2024.json descentralizacion-competencias.json; do
            curl "${CURL[@]}" "https://cepoes.org/assets/data/$name?$Q" | python -m json.tool >/dev/null
          done

          echo "✓ PRODUCCIÓN ACTUALIZADA Y VERIFICADA"

      - name: Restaurar producción anterior si falla publicación o smoke
        if: failure() && steps.publish.outcome == 'success'
        shell: bash

          set sftp:auto-confirm yes

          set net:max-retries 3
          set net:timeout 30
          mirror -R --verbose --parallel=2 --no-perms --no-umask _rollback "$REMOTE_DIR"
          bye
          EOF
          echo "✓ rollback completado"
